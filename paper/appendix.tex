\section{Extrapolation proofs}

\subsection{Linear derivative-error bounds}
\label{app:proof_linear}

\begin{proof}[Proof of \Cref{lem:error_linear}]
Put \(e_{\mathrm{out}}(x)=\pi(x)-\pi_{\mathrm{ref}}(x)\). By the fundamental theorem of calculus,
\[
|e_{\mathrm{out}}(b_+)-e_{\mathrm{out}}(b_-)|=\left|\int_{b_-}^{b_+}e_{\mathrm{out}}'(x)\,\mathrm d x\right|
\leq\int_{b_-}^{b_+}C_{\mathrm{der}}\,\mathrm d x
=C_{\mathrm{der}}(b_+-b_-).
\]
\end{proof}

\begin{proof}[Proof of \Cref{thm:ood_linear_nD}]
Parametrize the segment by \(\zeta(\upsilon)=x_0+\upsilon d\), where
\(d=x_{\mathrm{ood}}-x_0\) and \(\upsilon\in[0,1]\). For
\(e_{\mathrm{out}}=\pi-\pi_{\mathrm{ref}}\), the chain rule gives
\[
\left\|\frac{\mathrm d}{\mathrm d\upsilon}e_{\mathrm{out}}(\zeta(\upsilon))\right\|
=\|\mathrm{D}e_{\mathrm{out}}(\zeta(\upsilon))d\|
\leq C_{\mathrm{der}}\|d\|.
\]
Integrating and adding \(\|e_{\mathrm{out}}(x_0)\|\leq\delta\) yields
\(\|e_{\mathrm{out}}(x_{\mathrm{ood}})\|\leq\delta+C_{\mathrm{der}}\|d\|\).
\end{proof}

\subsection{Scalar quadratic bound}
\label{app:proof_scalar}

\begin{proof}[Proof of \Cref{thm:ood_1d}]
With \(e_{\mathrm{out}}=\pi-\pi_{\mathrm{ref}}\), integrate the assumed envelope:
\[
|e_{\mathrm{out}}(b_+)|\leq|e_{\mathrm{out}}(b_-)|+\int_{b_-}^{b_+}|e_{\mathrm{out}}'(x)|\,\mathrm d x
\leq\delta+\int_{b_-}^{b_+}
[\varepsilon+L_{\mathrm{err}}(x-b_-)]\,\mathrm d x.
\]
The integral equals
\(\varepsilon(b_+-b_-)+L_{\mathrm{err}}(b_+-b_-)^2/2\).
\end{proof}

\subsection{Path-envelope integration}
\label{app:proof_path}

\begin{proof}[Proof of \Cref{lem:path_fencing}]
The path error is continuously differentiable. Therefore
\[
\begin{split}
\|e_{\mathrm{path}}(1)\|
&\leq\|e_{\mathrm{path}}(0)\|
+\int_0^1\|e_{\mathrm{path}}'(\upsilon)\|\,\mathrm d\upsilon\\
&\leq\delta+\int_0^1
[\varepsilon\|d\|+L_{\mathrm{err}}\|d\|^2\upsilon]\,\mathrm d\upsilon\\
&=\delta+\varepsilon\|d\|+\tfrac12 L_{\mathrm{err}}\|d\|^2.
\end{split}
\]
\end{proof}

\subsection{Derivative-error propagation}
\label{app:component_lemma}

\begin{proof}[Expanded proof of \Cref{prop:composition}]
Insert the boundary derivatives and apply the triangle inequality:
\[
\begin{aligned}
\|\mathrm D\pi(x)-\mathrm D\pi_{\mathrm{ref}}(x)\|
&\leq \|\mathrm D\pi(x)-\mathrm D\pi(x_0)\|
+\|\mathrm D\pi(x_0)-\mathrm D\pi_{\mathrm{ref}}(x_0)\|\\
&\quad+\|\mathrm D\pi_{\mathrm{ref}}(x_0)-\mathrm D\pi_{\mathrm{ref}}(x)\|\\
&\leq L_\pi\|x-x_0\|+\varepsilon+M\|x-x_0\|.
\end{aligned}
\]
This is \eqref{eq:deriv_error_propagation} with
\(L_{\mathrm{err}}=L_\pi+M\).
\end{proof}

\subsection{Whole-policy quadratic bound}
\label{app:proof_main}

\begin{proof}[Proof of \Cref{thm:ood_quad_nD}]
By \Cref{prop:composition}, at every point of the segment,
\[
\|\mathrm{D}\pi(\zeta(\upsilon))
-\mathrm{D}\pi_{\mathrm{ref}}(\zeta(\upsilon))\|
\leq\varepsilon+L_{\mathrm{err}}\upsilon\|d\|.
\]
Multiplication by \(d\) through the chain rule bounds the path-error derivative by
\(\varepsilon\|d\|+L_{\mathrm{err}}\upsilon\|d\|^2\).
The base error is at most \(\delta\). Apply \Cref{lem:path_fencing}.
For \(d=0\), the statement is simply the base-error assumption.
\end{proof}

\section{Bellman and symbolic-reference proofs}

\subsection{Frozen-operator contraction}
\label{app:proof_contraction}

\begin{proof}[Proof of \Cref{thm:csbapr_contraction}]
Let \(d_Q=\|Q_1-Q_2\|_\infty\). The actor and temperature are identical in the two soft values, so their entropy terms cancel:
\[
|V^{Q_1}(x')-V^{Q_2}(x')|
\leq\int_{\mathcal A}p_\theta(a'\mid x')
|Q_1(x',a')-Q_2(x',a')|\,\mathrm d a'\leq d_Q.
\]
Rewards and frozen penalties also cancel. Since each \(P_\ell\) is a probability kernel,
\[
|(\mathcal T^{\mathrm{CS}}_\ell Q_1)(x,a)
-(\mathcal T^{\mathrm{CS}}_\ell Q_2)(x,a)|
\leq\gamma\int_{\mathcal S}d_Q P_\ell(dx'\mid x,a)=\gamma d_Q.
\]
Averaging with the fixed nonnegative weights \(\rho(\ell)\), whose sum is one, and taking the supremum preserves this inequality. Bounded rewards, entropy contributions, and penalties make the operator a self-map on the complete space of bounded measurable value functions. The contraction mapping theorem gives the unique fixed point.
\end{proof}

\subsection{Belief normalization}
\label{app:proof_belief}

\begin{proof}[Proof of \Cref{cor:belief_update}]
Every \(\rho^+(\ell)\) is nonnegative and
\(\sum_\ell\rho^+(\ell)=Z_\rho/Z_\rho=1\).
Appendix~\ref{app:proof_contraction} therefore applies with \(\rho^+\) fixed.
The conclusion concerns this new operator, not convergence of a sequence of changing operators.
\end{proof}

\subsection{Value-dependent feedback}
\label{app:proof_feedback}

\begin{proof}[Proof of \Cref{thm:frozen_sym_necessary}]
In the one-state, one-action example the entropy is zero and the next-state value is \(Q\). Absorbing all fixed terms into a constant \(C_0\), the backup is
\[
\mathcal T Q=C_0+\gamma(1+\lambda_{\mathrm{sym}}\nu)Q.
\]
For distinct scalar values its exact Lipschitz ratio is
\(\gamma(1+\lambda_{\mathrm{sym}}\nu)\).
A sufficiently large \(C_{\mathrm{sym}}\) makes
\(C_{\mathrm{sym}}-\nu Q\geq0\) on any chosen bounded comparison interval.
The affine map contracts precisely when its ratio is below one.
\end{proof}

\subsection{Symbolic-reference pipeline}
\label{app:proof_symbolic}

\begin{proof}[Proof of \Cref{thm:sindy_jac,thm:sindy_pipeline}]
At any point where the two hypotheses hold,
\[
\begin{split}
\|\mathrm D\pi-\mathrm D\pi_{\mathrm{ref}}\|
&\leq\|\mathrm D\pi-\mathrm D\pi_{\mathrm{sym}}\|
+\|\mathrm D\pi_{\mathrm{sym}}-\mathrm D\pi_{\mathrm{ref}}\|\\
&\leq\varepsilon_2+\varepsilon_1.
\end{split}
\]
If this holds throughout the segment, apply \Cref{thm:ood_linear_nD}
with \(C_{\mathrm{der}}=\varepsilon_1+\varepsilon_2\).
If it holds only at \(x_0\), it bounds the boundary mismatch;
\Cref{prop:composition,thm:ood_quad_nD} then require and use the whole-segment regularity assumptions.
Both derivatives in the argument are action-valued, after the stated control construction; a state-valued dynamics output is not interchangeable with that reference.
\end{proof}

\subsection{Conditional invariant filtering}
\label{app:proof_irm}

\begin{proof}[Proof of \Cref{thm:irm_causal}]
For the invariant model, decompose its test-environment derivative error as
\[
\mathrm D\pi_{\mathrm{ref}}^{e_\star}
-\mathrm D\pi_{\mathrm{sym}}^{\mathrm{inv}}
=(\mathrm D\pi_{\mathrm{inv}}
-\mathrm D\pi_{\mathrm{sym}}^{\mathrm{inv}})
+\mathrm D\pi_{\mathrm{var}}^{e_\star}.
\]
Its norm is at most
\(\varepsilon_{\mathrm{inv}}+\varepsilon_{\mathrm{var}}^{e_\star}\).
For the single-environment model insert the training-environment reference:
\[
\begin{split}
\mathrm D\pi_{\mathrm{ref}}^{e_\star}
-\mathrm D\pi_{\mathrm{sym}}^{\mathrm{single}}
={}&[\mathrm D(\pi_{\mathrm{inv}}+\pi_{\mathrm{var}}^{e_0})
-\mathrm D\pi_{\mathrm{sym}}^{\mathrm{single}}]\\
&+\mathrm D\pi_{\mathrm{var}}^{e_\star}
-\mathrm D\pi_{\mathrm{var}}^{e_0}.
\end{split}
\]
The triangle inequality gives the stated second bound. These are upper bounds on two distinct errors. Their difference does not establish a lower bound on the difference of the actual errors.
\end{proof}

\begin{proof}[Proof of \Cref{thm:irm_pipeline}]
In environment \(e_\star\), add the uniform derivative bounds
\(\varepsilon_{\mathrm{irm}}\) and \(\varepsilon_2\) by the same triangle inequality.
Apply \Cref{thm:ood_linear_nD} with the stated base error.
\end{proof}

\section{Implemented update equations}
\label{app:operator_detail}

For a replay transition \((x,a,r,x',d_{\mathrm{term}})\),
\(d_{\mathrm{term}}\in\{0,1\}\) is the terminal indicator and \(\widetilde r\)
is the normalized reward when normalization is enabled. Sample
\(a'\sim p_\theta(\cdot\mid x')\).
Let \(n_{\bar\phi_k}\geq0\) be the parameter norm returned by
\texttt{compute\_reg\_norm}: the sum of absolute target-critic weights and
hidden biases, excluding the output bias. The code adds this norm offset;
it is not a directly measured aleatoric uncertainty.
For the minimum-ensemble setting, the detached target for critic \(k\) is
\begin{equation}
\begin{split}
y_k=\widetilde r+\gamma(1-d_{\mathrm{term}})\big[
&Q_{\min}(x',a')-\alpha\log p_\theta(a'\mid x')\\
&+\lambda_{\mathrm{reg}}n_{\bar\phi_k}
-\lambda_{\mathrm{sym}}\widehat{\mathcal L}_{\mathrm{JC}}\big],
\end{split}
\label{eq:critic_target}
\end{equation}
where \(\lambda_{\mathrm{reg}}\) is the norm-offset coefficient and
\(\widehat{\mathcal L}_{\mathrm{JC}}\) is the batch estimator of
\eqref{eq:jac_loss} in the revised implementation. The historical symbolic
target instead used \eqref{eq:projected_loss}. For the per-critic-target setting,
\(Q_{\min}\) is replaced by \(Q_{\bar\phi_k}\).
The symbolic term is omitted when the symbolic model is unavailable or disabled.

Define the ensemble mean and sample standard deviation by
\[
\bar Q(x,a)=\frac{1}{N_{\mathrm{crit}}}\sum_{k=1}^{N_{\mathrm{crit}}}Q_{\phi_k}(x,a),
\qquad
\sigma_Q(x,a)=
\left[\frac{\sum_k(Q_{\phi_k}(x,a)-\bar Q(x,a))^2}
{N_{\mathrm{crit}}-1}\right]^{1/2}.
\]
The shared-critic loss is
\begin{equation}
\mathcal L_Q=
\mathbb E_{\mathrm{batch}}\!\left[
\frac{1}{N_{\mathrm{crit}}}\sum_k(Q_{\phi_k}(x,a)-y_k)^2
+\beta_{\mathrm{ood}}\sigma_Q(x,a)\right],
\label{eq:critic_loss}
\end{equation}
where \(\mathbb E_{\mathrm{batch}}\) denotes the replay-batch average and
\(\beta_{\mathrm{ood}}\) is its spread coefficient.
For a fresh action \(\widehat a\sim p_\theta(\cdot\mid x)\), the arithmetic-actor objective is
\begin{equation}
\begin{split}
\mathcal L_\pi={}&
\mathbb E_{\mathrm{batch},\,\widehat a}
[-\bar Q(x,\widehat a)-\beta_{\mathrm{eff}}\sigma_Q(x,\widehat a)
+\alpha\log p_\theta(\widehat a\mid x)
+\frac{\beta_{\mathrm{bc}}}{d_a}\|\widehat a-a\|^2]\\
&+\lambda_{\mathrm{JC}}\widehat{\mathcal L}_{\mathrm{JC}}
+\lambda_{\mathrm{NAU}}\mathcal L_{\mathrm{NAU}}.
\end{split}
\label{eq:actor_loss}
\end{equation}
Here \(\lambda_{\mathrm{JC}}\) and \(\lambda_{\mathrm{NAU}}\) weight the
full action-space consistency term and the NAU weight regularizer. The revised
actor uses the unclipped consistency loss; the historical version clipped
the projected loss at 2. The implemented
BC term is a mean squared error over the \(d_a\) action coordinates.
Other heads omit the NAU regularizer or use their own regularizer.
The primary transit comparisons set \(\lambda_{\mathrm{JC}}\) and
\(\lambda_{\mathrm{sym}}\) to zero. The six-setting comparison does not
activate the optional Q-spread caps or extra curvature penalties.

For the implementation's run-length hypotheses, \(\ell\) takes the values
\(0,\ldots,H_{\max}-1\). At update \(j\), the code uses
\[
\lambda_w=\sum_{\ell=0}^{H_{\max}-1}\rho_j(\ell)\exp(-0.1\ell)
\]
after warmup; it is zero during warmup. Here \(\rho_j\) is the belief at
update \(j\), and \(H_{\max}\) is the maximum
run-length hypothesis count. The actor coefficient is
\(\beta_{\mathrm{eff}}=-2-c_{\mathrm{pen}}\lambda_w\).
The norm offset, sampled next actions, target network, actor, and symbolic
model are detached or held fixed for each critic target construction.
The abstract contraction result concerns a fixed actor and fixed penalty
fields; it does not establish convergence of these changing gradient updates.

\section{Symbolic fitting details}
\label{app:sindy_details}

The configured polynomial library is fit with STLSQ. Write
\(\mathsf X\) for joint sampled state-action inputs and \(\mathsf Y\) for regression targets.
In discrete mode the implemented target is the state increment
\(\Delta x_t=x_{t+1}-x_t\), not the next state itself:
\[
\mathsf Y\approx\boldsymbol\Phi(\mathsf X)\Xi^{\mathsf T},
\]
where the superscript \(\mathsf T\) denotes matrix transpose.
In continuous mode finite differences approximate the state derivative
using the sampling interval \(\Delta t\), changing the regression target
and its units. The frozen Torch wrapper evaluates
the polynomial terms and fitted coefficient matrix \(\Xi\), giving
\(f_{\mathrm{sym}}\) under the paper's discrete-mode convention.
The revised fit aligns every \(a_t\) with \(x_t\) and its increment before
concatenating episodes. Its wrapper uses the fitted polynomial library's
exponent matrix, including state-action cross terms and configured degrees
up to four. Both prediction paths require actions for control-aware models.
For degree-one, zero-intercept fits, the state coefficients describe
\(\widehat A-I\), and the action coefficients describe
\(\widehat B_{\mathrm{in}}\).
In LQR the environment applies force \(5a\), so the fitted input matrix
is in normalized-action units and the cost matrix is
\(R_{\mathrm{cost}}=25\operatorname{diag}(0.01,0.01)\);
\(Q_{\mathrm{cost}}=\operatorname{diag}(1,1,0.1,0.1)\).
The frozen reference uses the identified matrices with these task costs.
Prediction-quality diagnostics use the coefficient of determination
\(R^2\); they do not measure a uniform derivative error.
A prediction-quality threshold in the configuration is not a certificate
of \Cref{thm:sindy_jac}'s action-reference hypotheses.

\section{Arithmetic-head implementation}
\label{app:nau_details}

The historical feature extractor consists of two affine layers with LeakyReLU slope
\(0.01\), denoted \(G\). A fixed linear projection \(U_{\mathrm{quad}}\)
feeds the elementwise quadratic correction. The pre-squash mean is
\[
\mu_\theta(x)=
\eta_{\mathrm{mix}}W_{\mathrm{NAU}}G(x)
+(1-\eta_{\mathrm{mix}})
c_{\mathrm{quad}}\odot(U_{\mathrm{quad}}G(x))^{\odot2},
\]
where \(\odot\) denotes coordinatewise multiplication,
\(c_{\mathrm{quad}}\) is the vector of quadratic coefficients, and
\(\eta_{\mathrm{mix}}\in(0,1)\) is a learned sigmoid-transformed mixing
weight. Its unconstrained parameter is initialized at \(0.8\), giving
\(\eta_{\mathrm{mix}}=(1+\exp(-0.8))^{-1}\approx0.690\).
A separate head predicts the Gaussian log standard deviation.
For normalized actions the deterministic policy is \(\pi=\tanh\mu_\theta\).
An affine physical rescaling has offset \(b_a\) and scale \(c_a>0\);
in transit, \(h_{\mathrm{hold}}=b_a+c_a\pi\) with \(b_a=c_a=30\) s.

Training clamps \(W_{\mathrm{NAU}}\) to \([-1,1]\) and uses
\begin{equation}
\mathcal L_{\mathrm{NAU}}
=\frac{1}{N_W}\sum_{u,v}
\min\bigl(|(W_{\mathrm{NAU}})_{uv}|,\,
1-|(W_{\mathrm{NAU}})_{uv}|\bigr),
\label{eq:nau_regularizer}
\end{equation}
where \(u,v\) index matrix rows and columns and \(N_W\) is the number of
entries. This loss encourages discrete values but does not require exact
discreteness for the NAU derivative identity.
The regularizer in \eqref{eq:nau_regularizer} is the term used in
\eqref{eq:actor_loss}. The quadratic correction remains active in
\texttt{csbapr}, so this configuration is not a pure linear NAU actor.

\subsection{Complete composition calculation}
\label{app:leff}

Let \(G\) and \(H\) be continuously differentiable on neighborhoods of
the convex domains under consideration. Suppose
\(\|\mathrm DG\|\leq K_G\), \(\|\mathrm DH\|\leq K_H\), and their
derivatives have Lipschitz constants \(L_G,L_H\), respectively.
The derivative chain rule gives
\[
\begin{split}
\mathrm D(H\circ G)(x)-\mathrm D(H\circ G)(y)
={}&[\mathrm DH(G(x))-\mathrm DH(G(y))]\mathrm DG(x)\\
&+\mathrm DH(G(y))[\mathrm DG(x)-\mathrm DG(y)].
\end{split}
\]
Since \(\|G(x)-G(y)\|\leq K_G\|x-y\|\), its norm is at most
\((L_HK_G^2+K_HL_G)\|x-y\|\), proving
\eqref{eq:composition_rule}.
For the mixed head, a valid feature-space derivative Lipschitz bound is
\[
L_{H_{\mathrm{mix}}}
\leq2(1-\eta_{\mathrm{mix}})
\|c_{\mathrm{quad}}\|_\infty\|U_{\mathrm{quad}}\|_{\mathrm{op}}^2.
\]
Here the infinity norm is the maximum absolute vector entry. Even when the
head is linear, the \(K_HL_G\) term does not vanish unless the features
have constant derivatives.

For componentwise action squashing,
\(\sup|\tanh'|=1\) and
\(\sup|\tanh''|=4/(3\sqrt3)\).
One more application of the composition calculation yields
\begin{equation}
L_\pi\leq
\frac{4}{3\sqrt3}(K_HK_G)^2+L_HK_G^2+K_HL_G.
\label{eq:complete_composition}
\end{equation}
Equation~\eqref{eq:complete_composition} bounds the normalized deterministic
action map, including both feature curvature and tanh curvature.
An affine physical rescaling multiplies this derivative bound by \(c_a\).
The historical logged curvature surrogate omitted these global terms.
For example, \(\pi(x)=\tanh(\operatorname{LeakyReLU}_{0.01}(x))\)
has left and right derivatives \(0.01\) and \(1\) at zero. No finite
global derivative Lipschitz constant exists across that activation boundary.
Within a fixed activation region the feature map is affine, but such a
local calculation does not certify a segment that crosses regions.

\paragraph{Smooth arithmetic variant.}
The revised \texttt{smooth-nau} actor replaces both feature activations by
tanh, keeping the arithmetic and quadratic heads. Let \(W_1,W_2\) be the
two feature-layer matrices; their common hidden width is the feature
dimension \(d_z\). Define \(c_{\tanh}=4/(3\sqrt3)\). Since componentwise tanh has
derivative norm at most one and derivative Lipschitz constant
\(c_{\tanh}\), two applications of \eqref{eq:composition_rule} give
\[
\begin{aligned}
K_G&\leq\|W_2\|_{\mathrm{op}}\|W_1\|_{\mathrm{op}},\\
L_G&\leq c_{\tanh}\big[(\|W_2\|_{\mathrm{op}}\|W_1\|_{\mathrm{op}})^2
+\|W_2\|_{\mathrm{op}}\|W_1\|_{\mathrm{op}}^2\big].
\end{aligned}
\]
The features obey \(\|G(x)\|\leq R_G=\sqrt{d_z}\), where
\(R_G\) is their global radius bound. Thus
\[
K_H\leq\eta_{\mathrm{mix}}\|W_{\mathrm{NAU}}\|_{\mathrm{op}}+L_HR_G.
\]
Substituting these constants and the quadratic-head bound into
\eqref{eq:complete_composition} yields the implemented global certificate.
Feature biases do not affect the norm bounds because tanh remains bounded.
For \(\pi_{\mathrm{sym}}(x)=\tanh(-\widehat K_{\mathrm{fb}}x)\), the same
calculation gives reference curvature at most
\(c_{\tanh}\|\widehat K_{\mathrm{fb}}\|_{\mathrm{op}}^2\).
The boundary errors are computed as Euclidean action error and operator-norm
Jacobian error at \(x_0\), then inserted into \eqref{eq:ood_bound}.

\section{Cross-environment selection}
\label{app:irm_protocol}

For each fitted symbolic candidate \(\Xi\), let
\(\widehat E_e(\Xi)\) be its mean Euclidean prediction error in training
environment \(e\), evaluated on that environment's collected states and
targets. The implementation selects
\begin{equation}
\Xi_{\mathrm{sel}}\in
\operatorname*{arg\,min}_{\Xi\in\mathfrak C}
\operatorname{Var}_{e\in\mathcal E_{\mathrm{train}}}
[\widehat E_e(\Xi)],
\label{eq:invariance_selector}
\end{equation}
where \(\mathfrak C\) is the finite set of fitted candidates and
\(\Xi_{\mathrm{sel}}\) is the selected coefficient matrix.
The variance criterion \eqref{eq:invariance_selector} differs from
the earlier coefficient-averaging implementation as well as
IRMv1's gradient penalty. It can favor a model with uniformly large
prediction error. Neither derivative accuracy nor causal identification
follows from low variance alone. The default uses three environments;
their required number and diversity depend on the identification problem.
